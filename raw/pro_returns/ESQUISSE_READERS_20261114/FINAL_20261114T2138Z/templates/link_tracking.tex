% Private PDF metadata identifies one logical URI link across its wrapped
% annotation rectangles. It changes no visible text, target or link geometry.
\newcounter{editionexternallink}
\makeatletter
\let\EditionBaseLinkURL\hyper@linkurl
\let\EditionBasePDFBorder\Hy@setpdfborder
\def\hyper@linkurl#1#2{%
  \begingroup
  \stepcounter{editionexternallink}%
  \edef\EditionLinkIdentifier{edlink-\theeditionexternallink}%
  \def\Hy@setpdfborder{\EditionBasePDFBorder /EsquisseLinkID (\EditionLinkIdentifier)}%
  \EditionBaseLinkURL{#1}{#2}%
  \endgroup}
\makeatother
