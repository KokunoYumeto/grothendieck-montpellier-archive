\subsection{The same coverage in arbitrary even weight}
\label{sec:u3-general-coverage}

\begin{corollary}[K17-U3-GENERAL-COVERAGE]
\label{cor:u3-general-coverage}
Let $\ell$ be an odd prime, let $e\ge2$ be even, put
$E=\mathbf Q(\sqrt{-3})$, and retain the standard
positive definite $U(3)$ and the full lattice level $K$ of
Theorem~\ref{thm:u3-coverage}. Every representation in the actual source
predicate $\mathrm{Adm3U\_s}(\iota,e,2)$ is attached to an everywhere
finite-unramified, regular algebraic conjugate-self-dual cuspidal
$\mathrm{GL}_3(\mathbf A_E)$ representation with unitary infinity
parameter
\[
 z\longmapsto
 \operatorname{diag}\bigl((z/\bar z)^e,1,(z/\bar z)^{-e}\bigr).
\]
That representation occurs, with its split-prime Hecke system, in the
full algebraic modular-form space for $K$ and compact coefficient
highest weight $(e-1,0,1-e)$.

For a split prime $p\ne\ell$, let $\alpha$ be its eigenvalue for the
scaled compact Hecke operator $p^{e-1}T_p$, where $T_p$ is the positive
minuscule operator. For arithmetic Frobenius $\sigma$,
\[
 \iota\operatorname{tr}\rho(\sigma^{-1})=\alpha/p^e.
 \tag{U1}
\]
All the source's strengthened $\mathrm{WitA}$, $\mathrm{Th.Unl}$, and
cyclotomic-twist hypotheses are retained in this assertion.
\end{corollary}

\begin{proof}
We spell out why the preceding proofs are uniform in $e$.
The exact definitions in \texttt{NumberU3US.lean},
\texttt{AfStmt.lean}, and \texttt{IsAutCsdA.lean} give
irreducibility, $\mathrm{Level}(\ell,2,(e,0,-e),\rho)$,
$r=\rho\chi_\ell^{-1}$ with $\mathrm{AutCsd0}(\iota,r)$, the implication
from continuity of $r$ to its actual $\mathrm{WitA}$, and the additional
$\mathrm{Th.Unl}$.
As in Lemma~\ref{lem:step17-u3-level}, every
$\mathrm{KillsMu}\,2$ character is trivial since $\mu_2\subset\mathbf Q$.
The same witnesses therefore give
$\mathrm{Level}(\ell,1,(e,0,-e),\rho)$.
The proved cyclotomic-twist theorem gives
\[
 \mathrm{Level}(\ell,1,(e-1,-1,-e-1),r).
 \tag{U2}
\]
This includes continuity, so $\mathrm{WitA}(\iota,r)$ is available.
The full theorem \texttt{unlAll\_of} gives each above-$\ell$ flag for
$\rho$ with $\kappa_i\ne\kappa_j\chi_\ell$. A common
$\chi_\ell^{-1}$ twist gives the same condition for $r$.
Apply \texttt{Keep.levelKA\_alt}, or its complete
\texttt{levelCore\_alt} proof, at modulus one to (U2).
The result is a $K\mathrm{Lev}(E,3,1)$ fixed vector, exact
Frobenius--Satake compatibility at every place away from $\ell$, and
\[
 \operatorname{InfChar}_\tau(\Pi)
 =\operatorname{infOf}(e-1,-1,-e-1)=(e,0,-e).
 \tag{U3}
\]
Modulus one is the product of all local
$\mathrm{GL}_3(O_{E_w})$, including above $\ell$.
The exponent vector in (U2) is strictly decreasing for $e>0$,
and the Hodge--Tate weights are $(1-e,1,1+e)$.
Thus the source weight relation
$\mathrm{HT}=1-\operatorname{InfChar}$ has exactly the stated
normalization.

For the compact transfer, repeat the proof of
Theorem~\ref{thm:u3-coverage}, replacing $(6,0,-6)$ by $(e,0,-e)$.
Every input remains valid: the three distinct infinity characters are
conjugate-orthogonal because their exponent differences are even; the
global simple generic parameter has trivial reduced component group;
and the compact Hermitian group is a pure inner form.
The infinity member has highest weight
$(e,0,-e)-(1,0,-1)=(e-1,0,1-e)$.
Shin's theorem applies in all these regular algebraic odd-rank cases,
so every finite parameter is bounded.
The split and inert spherical choices are unchanged.
At the ramified prime three, the central character remains trivial:
it is unramified everywhere finite, trivial at infinity, and $E$ has
class number one.
The unramified local base-change roots are consequently
$\{a,1,a^{-1}\}$, with unitary $a$.
The same integral identification with the Adler--Lansky lattice, the
same irreducible or quadratic-reducible principal-series cases, and
the same exactness of full lattice-stabilizer invariants give a local
packet member fixed by $K_3$. None of these arguments uses the value
six. The proved generic pure-inner multiplicity formula therefore
gives the asserted global member with multiplicity one.
The primary locators and the depth-preservation-hypothesis bypass are
exactly those in the complete proof of
Theorem~\ref{thm:u3-coverage}.

Finally, the source twist always satisfies
$r(\sigma^{-1})=p\rho(\sigma^{-1})$, and its
$\operatorname{algSat}(3,p)$ multiplies the unitary roots by $p$.
Thus $\operatorname{tr}\rho(\sigma^{-1})$ equals their sum.
The normalized Satake formula for the positive minuscule operator of
$\mathrm{GL}_3$ is $T_p=p\sum\beta$.
Multiplication by $p^{e-1}$ proves (U1).
If a finite model is computed with inverse minuscule representatives,
its independently checked conjugation similarity identifies the
characteristic polynomial with that of this positive operator.
\end{proof}

\begin{corollary}[The general finite-polynomial unit criterion]
\label{cor:general-hecke-unit}
Assume the source datum $\mathrm{GD}(\Omega,E,\ell,e,1)$ and the
rank-two conductor and trace bridge for the same $e$.
Let $P_{e,p}$ be the characteristic polynomial of the entire scaled
compact coefficient space in Corollary~\ref{cor:u3-general-coverage},
and suppose $P_{e,p}\in\mathbf Z[X]$.
Let $Q_{e,p}$ be the characteristic polynomial on the entire
$S_{2e+1}(\Gamma_0(3),\chi_{-3})$.
Put
\[
 N_{e,p}=P_{e,p}(1+p^e+p^{2e})Q_{e,p}(1+p^{2e}).
\]
If $p$ is split in $E$, $p\ne\ell$, and $\ell\nmid N_{e,p}$, then the
source negative-character Selmer module has length zero.
\end{corollary}

\begin{proof}
At $s=\sigma^{-1}$, the source comparison trace is
$p^{-e}+1+p^e$.
By Corollary~\ref{cor:u3-general-coverage}, every admitted rank-three
trace difference is
\[
 \bigl(\alpha-(1+p^e+p^{2e})\bigr)/p^e
\]
for a root $\alpha$ of $P_{e,p}$.
The rank-two bridge puts every other admitted trace difference in
\[
 \bigl(a_p-(1+p^{2e})\bigr)/p^e
\]
for a root $a_p$ of $Q_{e,p}$.
Both polynomials are monic and integral. At every place above $\ell$,
their roots are integral, and a polynomial evaluation which is a unit
is a product of integral factors and therefore has each factor a unit.
Since $p\ne\ell$, both displayed denominators are units as well.
Lemma~\ref{lem:step17-unit} applies to every actual admitted
representation. This argument requires pointwise coverage only; no
comparison of indexing sets or characteristic-polynomial
multiplicities with the source's \texttt{TrData} is used.
\end{proof}
