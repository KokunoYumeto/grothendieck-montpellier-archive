\margin{(29.11.83)}

Let $X$ be a compact oriented surface with boundary, with toric structure
\inserted{(that is, torsor under $\mathbb U=\{z\in\mathbb C\mid |z|=1\}$)}
given on the connected components of the boundary.
We define
\[
\begin{aligned}
\operatorname{SRT}(\unsure{X})={}&
\text{quotient of the group of automorphisms of }X
\text{ (\deleted{\unsure{in}} preserving the orientation}\\
&\text{and the toric structures) by the identity component of the}\\
&\text{subgroup }\operatorname{Aut}^!(\vec X)
\deleted{\unsure{\mathrm{bijective}}}
\text{ of Aut which induces}\\
&\text{the identity on }\partial X
\end{aligned}
\]

We thus have a \unsure{dévissage} of $\operatorname{SRT}(X)$ in three stages
\[
\underbrace{S_{\unsure{*}}T^!(X)}_{
\operatorname{Aut}^!(\unsure{\vec X|\overline X})/
\text{identity component}}
\qquad\mathbb U^I\qquad\unsure{\mathfrak S_I}
\qquad I=\pi_0(\partial X)
\]
or again
\[
\mathbb Z^I\qquad T(X)\qquad\mathbb U^I\qquad\unsure{\mathfrak S_I}
\]

In fact, we define canonically! a \inserted{straight} subgroup
$\mathbb R^I$ of $\operatorname{SRT}(X)$
containing $\mathbb Z^I$, in such a way that we have
\[
\underbrace{\operatorname{SRT}^!(X)}_{
\substack{\text{kernel of}\\
\operatorname{SRT}^!(\unsure{\alpha|X})\to\unsure{\mathfrak S_I^!|\mathfrak S_I'}}}
=\mathbb R^I.S_*T^!(X)
\]

\diagram{
\begin{tikzcd}[column sep=small,row sep=large]
\mathbb R^I &
\mathord{:}\arrow[l,phantom,"\subset"']\arrow[dr,no head,"\unsure{\lambda}"] &&\\
\mathbb Z^I\arrow[u,"\text{cocartesian}"']\arrow[r,hook]&
S_*T^!\arrow[u,hook]\arrow[r,phantom,"\subset"]&
\operatorname{SRT}^!\arrow[r,phantom,"\subset"]&
\operatorname{SRT}
\end{tikzcd}
}
\margin{\deleted{\unsure{and proper}}}

that is, we have

\margin{\uline{Note well} $\operatorname{SUT}(X)\simeq
T(X)\times_{\unsure{\mathfrak S_I}}\operatorname{Aut}((T_i)_{i\in I})$}

\margin{\unsure{so far we bring in}
only $S_*T^!(X)$; to the exclusion of $S_*T(X)$,
whose definition requires an additional structure
$\uline{s}$, see below}

\diagram{
\begin{tikzcd}[column sep=small,row sep=large]
&\operatorname{SRT}(X)\arrow[d,no head,"\unsure{\mathfrak S_I}"]&\\
&\mathbb R^I.\unsure{S_*T^!}\arrow[dl,no head,"T^!"']
 \arrow[dr,no head,"\mathbb U^I"]&\\
\mathbb R^I\arrow[dr,no head,"\mathbb U^I"']&&S_*T^!(X)\\
&\mathbb Z^I\arrow[ur,no head,"T^!(X)"']&
\end{tikzcd}
}
\margin{$T(X)\left\{\vphantom{\begin{matrix}a\\a\\a\end{matrix}}\right.$}
\margin{$\left.\vphantom{\begin{matrix}a\\a\\a\end{matrix}}\right\}\operatorname{Aut}((T_i)_{i\in I})$}

whence
\diagram{
\begin{tikzcd}[row sep=large,column sep=small]
\operatorname{SRT}\deleted{(X)}/\mathbb Z^I
 \overset{\mathrm{def}}=\operatorname{SUT}(X)
 \arrow[d,"\unsure{\mathfrak S_I}"]\\
T^!(X)\times\mathbb U^I
\end{tikzcd}
}
\[
(*)\qquad1\longrightarrow\mathbb U^I\longrightarrow
\operatorname{SUT}(X)\longrightarrow T(X)-1
\]

Moreover, each time we give ourselves a
marking, that is, a point \deleted{\unsure{of}}
\[
\uline{s}=(s_i)\in
\prod_{i\in I=\pi_0(\partial X)}
\deleted{\unsure{\mathbb U_i}}T_i
\]
\margin{$T_i=$ connected component (toric) of $\partial X$ with index $i$}

we find an embedding
\[
S_*T^{\deleted{\unsure{!}}}(X,\uline{s})\hookrightarrow\operatorname{SRT}(X)
\]
such that we deduce from it
\[
S_*T^{\deleted{\unsure{!}}}(X,\uline{s})
\mathbin{\wedge}_{\mathbb Z^I}\mathbb R^I
\simeq\operatorname{SRT}(X)
\]
