Recapitulation of the elementary operations

\inserted{$(0^\circ)$ \unsure{Disjoint|Direct} sum}

$1^\circ$) \uline{erasing} \inserted{\uline{all} the marks at $a$, that is} the marks lying on a circle \inserted{\unsure{$C_a$}} for gluing \inserted{\unsure{the|$C_a$}}, or \deleted{\unsure{on a circle}} for cutting
$\Longleftrightarrow$ removal of the \deleted{part} \inserted{$R_a$}
$(a\in A)$ $\Longleftrightarrow$ modification of $r_a$ by
\deleted{\unsure{of $R$ lying in a part}}
replacing it by $0$ at $a$.
\margin{\uline{strongly irreversible}\quad $r_a\ne0$}

$2^\circ$) \inserted{\uline{uniform partial erasure}: \uline{of exponent $d_a\mid r_a$}}
Restriction of the \deleted{scalars} \inserted{groups of $R_a$} from $\mathbb Z_a/r_a\mathbb Z$ to a
$\mathbb Z_a/r'_a\mathbb Z_a$, where $r'_a\mid r_a$
(for fixed $a$, there are $r_a/r'_a=d_a$ possible choices, if one assumes $d_a>1$ that is $r'_a\ne r_a$).
\margin{\uline{weak strengthening of structure}\quad $r_a\unsure{\ge|>}1$
\[
r'_a\unsure{\mid|\ne}r_a\text{ and }r'_a\ne r_a
\qquad\text{that is }d_a=\frac{r_a}{r'_a}\ne1.
\]}

$3^\circ$) \deleted{Extension of the}
\inserted{\uline{Uniform overmarking} \uline{of exponent} $d_a>1$}
--- that is extension of the groups of $R_a$ from
$\mathbb Z_a/r_a\mathbb Z_a$ to $\mathbb Z_a/d_ar_a\mathbb Z_a$,
\margin{\uline{weak weakening of structure}\quad $r_a\ne0$}

\textcircled{$4^\circ$} erasure of cutting circle $C_a$ $(a\in A_0)$

\textcircled{$5^\circ$} filling in hole $C_a$ $(a\in A_l)$
\margin{\uline{strongly irreversible}\quad $\left\{\begin{array}{l}4^\circ\\5^\circ\end{array}\right.$}

$6^\circ$) Cutting along $C_a$ $(a\in A_0)$ \({}^*\)
\margin{strongly irreversible if $r_a=0$, \deleted{weak} reversible = anodyne if $r_a\ge1$.}

$7^\circ$) Gluing along $C_a,C_{a'}$
(with $a,a'\in A_l$, $a\ne a'$, \inserted{$r_a=r_{a'}$}), via an element of
$R_a\mathbin{\wedge}R_{a'}$.
\margin{$r_a=r_{a'}\ge1$\quad reversible = anodyne.}

\margin{\({}^*\) One should \inserted{perhaps} split $6^\circ)$ into two, according as $r_a=0$ or $r_a\ne0$ (or exclude the case $r_a=0$?).}
