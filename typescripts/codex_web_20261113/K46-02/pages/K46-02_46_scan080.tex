\uline{Lemme 5} Soit
\diagram{\begin{tikzcd}[ampersand replacement=\&]
e\arrow[r,dash]\&\unsure{G'}\arrow[r,dash]\&G\arrow[r,dash]\&G''\arrow[r,dash]\&e

\[
