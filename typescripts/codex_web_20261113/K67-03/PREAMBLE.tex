  \newcommand{\deleted}{\ifmmode\expandafter\tr@deleted@math\else\expandafter\tr@deleted@text\fi}
  \def\tr@deleted@text{\bgroup\@strikeThrough\let\tr@next=}
  \newcommand{\tr@deleted@math}[1]{\text{\@strikeThrough$#1$}}
  % \uline and \uuline (ulem's names, used by the transcriptions) as lua-ul underlines in a group: a struck word
  % inside an underlined passage then compiles (ulem's word-by-word underline broke on it: K23-03, K79-11)

  \renewcommand{\uline}{\bgroup\@basicUnderLine\let\tr@next=}
  \renewcommand{\uuline}{\bgroup\@basicUnderLine\tr@lowerline\let\tr@next=}
