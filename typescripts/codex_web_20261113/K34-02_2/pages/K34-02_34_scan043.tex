\[
\pi_1(U,\xi)\longrightarrow\pi_1(S,\xi)\simeq\operatorname{Gal}(\overline k/k)

\[

\tag*{(8.5.3)}
\]

\[
I\simeq\pi_1(\widetilde U,\widetilde\xi),
\tag*{(8.5.4)}
\]
