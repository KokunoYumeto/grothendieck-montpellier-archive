\[
\begin{aligned}
 X\gtrless_{\mathrm{pol}}Y\ &\overset{\mathrm{def}}{\Longleftrightarrow}\ X\gtrless Y,\ \text{\uuline{et}}\ \forall X'\in\widetilde X,\ Y'\in\widetilde Y,\\
 &\widetilde{X'}\cap\widetilde{Y'}\text{ est vide ou a un}\\
