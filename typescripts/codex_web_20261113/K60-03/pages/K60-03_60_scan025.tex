\[
\mathbb R_\infty f(E)=\mathbb R\bar f_*j^*\underbrace{(\mathbb R i_*(E))}_{\big|}

\[
\begin{aligned}
D\mathbb R_\infty f(E)&=D\mathbb R\bar f_{\unsure{\infty|*}}(\hspace{3em})\\
&\left.\begin{aligned}
&=\mathbb R\bar f_*D(\hspace{3em})\\
